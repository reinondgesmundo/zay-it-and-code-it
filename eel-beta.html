def update():
    global x, y, applex, appley
    if keyboard.left:
        x -= 5
    if keyboard.right:
        x += 5
    if keyboard.up:
        y -= 5
    if keyboard.down:
        y += 5
    if x > 590:
        x = 590
    if x < 10:
        x = 10
    if y > 590:
        y = 590
    if y < 50:
        y = 50
    distance(x, y, applex, appley)
        
def distance(px, py, fx, fy):
    global applex, appley, score
    if abs(px - fx) < 25 and abs(py - fy) < 25:
        applex = random.randint(0, 600)
        appley = random.randint(50, 600)
        score += 1
