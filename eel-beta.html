import pgzrun
import random

WIDTH = 600
HEIGHT = 600
TITLE = "Eel Game Beta"

x = 300
y = 300
score = 0
applex = random.randint(0, 600)
appley = random.randint(50, 600)

length = 5      
GAP = 2        
trail = []     

for k in range(length * GAP, -1, -1):
    trail.append((x, y + k * 5))

def draw():
    screen.fill("lightblue")
    for i in range(length - 1, -1, -1):
        index = len(trail) - 1 - i * GAP
        if index >= 0:
            bx, by = trail[index]
            if i == 0:
                screen.draw.filled_rect(Rect(bx, by, 11, 11),  "darkblue")
            else:
                screen.draw.filled_rect(Rect(bx, by, 10, 10),  "blue")

    screen.draw.filled_circle((applex, appley), 5, "yellow")
    screen.draw.text("SCORE:" + str(score), (10, 10), fontsize=30, color="black")
    #print(applex, " ", appley)

def update():
    global x, y, applex, appley
    old_x = x
    old_y = y
    if keyboard.left:
        x -= 5
    if keyboard.right:
        x += 5
    if keyboard.up:
        y -= 5
    if keyboard.down:
        y += 5
    if x > 590:
        x = 590
    if x < 10:
        x = 10
    if y > 590:
        y = 590
    if y < 50:
        y = 50

  
    if x != old_x or y != old_y:
        trail.append((x, y))
        
        while len(trail) > length * GAP + 1:
            trail.pop(0)

    distance(x, y, applex, appley)

def distance(px, py, fx, fy):
    global applex, appley, score, length
    if abs(px - fx) < 25 and abs(py - fy) < 25:
        applex = random.randint(0, 600)
        appley = random.randint(50, 600)
        score += 1
        length += 1  

pgzrun.go()
