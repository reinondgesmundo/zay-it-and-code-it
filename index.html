PYTHON PROJECT #1

Ready to test your math skills?

Numbers range from 1 to 50. A correct answer gives you a point. After a wrong answer, choose whether to try again.

SCORE: 0

🐍 Zay’s original Python code

Preserved exactly as he wrote it. This code uses input(); the interactive game on the left is adapted for a webpage.

import random
score = 0
exit = "y"
while exit == "y":
    a = random.randint(1, 50)
    b = random.randint(1, 50)
    print("what is the sum of ", a, "and", b, "?")
    user = int(input(""))
    if user == a+b:
        print("You nailed it!")
        score += 1
    else:
        print("Wrong, the correct answer is", a+b)
        print("Want a another try?(y/n)")
        exit = input(" ")

    print("SCORE: ", score)
        
    
print("THANKS FOR PLAYING")

🐍 Zay’s original Python Turtle code

His code is shown below as written. The playable version uses JavaScript Canvas because desktop Python Turtle cannot run directly on GitHub Pages.

import turtle #a built-in graphics
import random #random numbers
import time #it will wait or check the date and time

screen = turtle.Screen() 
screen.setup(600, 600) #screen size
screen.bgcolor("lightgreen") #screen color
screen.tracer(0) #no screen updates

player = turtle.Turtle()
player.shape("square") #making a player using the shape
player.color("green") #player color
player.penup() #pen goes up
player.goto(0, -250) #coordinates

apple = turtle.Turtle()
apple.shape("circle")
apple.color("red")
apple.penup()
apple.goto(random.randint(-280, 280), random.randint(-280, 280))

chipbag = turtle.Turtle()
chipbag.shape("square")
chipbag.color("orange")
chipbag.penup()
chipbag.goto(random.randint(-280, 280), random.randint(-280, 280))

score = 0 #set a score
writer = turtle.Turtle()
writer.hideturtle() #hiding the pen
writer.penup()
writer.goto(-280, 260)
writer.write("Score: 0", font=("Arial", 16, "normal")) #writing the text

def go_left():
    player.setx(player.xcor() - 30) #sets the x - 30
    
def go_right():
    player.setx(player.xcor() + 30)
    
def go_up():
    player.sety(player.ycor() + 30)
    
def go_down():
    player.sety(player.ycor() - 30)
    
screen.listen() #it will focus on screen
screen.onkeypress(go_left, "Left")  #on key press
screen.onkeypress(go_right, "Right")
screen.onkeypress(go_up, "Up")
screen.onkeypress(go_down, "Down")

while True: #endless loop
    screen.update() #update the screen
    time.sleep(0.0000000001) #wait for 0.01 seconds
    
    
    
    if apple.distance(player) < 25: #when near
        score += 1 #add the score
        writer.clear() #remove his canvas
        writer.write("Scroe: " + str(score), font=("Arial", 16, "normal"))
        apple.goto(random.randint(-280, 280), random.randint(-280, 280)) #the apple appears in random location
        
        
    if chipbag.distance(player) < 25 and score >= 10: #when near
        score += 3 #add the score
        writer.clear() #remove his canvas
        writer.write("Scroe: " + str(score), font=("Arial", 16, "normal"))
        chipbag.goto(random.randint(-280, 280), random.randint(-280, 280)) #the apple appears in random location
